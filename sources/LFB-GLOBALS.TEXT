{ LFB-GLOBALS.TEXT -- LisaFuji Browser (tool 419314).                    }
{ A text web browser for the Lisa Office System.  The Lisa never fetches }
{ a web page itself: it asks a GATEWAY (a small Python service, on the   }
{ Banana now, on a Raspberry Pi later) for a page by address or by link  }
{ number.  The gateway fetches the page, keeps the words and the links,  }
{ numbers the links, wraps the text to 76 columns and hands it back 14   }
{ lines at a time, through the FujiNet on Serial B, in the same shape as }
{ the FujiNet news service.  All of the serial and FujiBus work is in    }
{ LFB-FNBUS (unit FnBus, LisaFuji's unit unchanged).  This unit is state }
{ only.                                                                   }

procedure HandleFilerOp (var wantsOpen : boolean);


IMPLEMENTATION

procedure HandleFilerOp {var wantsOpen : boolean};
var fx  : FilerExt;
    err : integer;
begin
   wantsOpen := false;
   GetAddParms(err, theEvent, fx);

   case fx.theFlrOp of

      fcNone :
         wantsOpen := true;              { the user opened the tool }

      fcResume :
         TellFiler(err, docClosd, internalError, theEvent.who);

      fcClose, fcSuspend, fcShred :
         begin
            myWindow := nil;
            TellFiler(err, docClosd, allOK, theEvent.who)
         end;

      fcCopy, fcPut :
         TellFiler(err, docNotXfered, internalError, theEvent.who);

      fcDfClose :
         TellFiler(err, dfClosed, allOK, theEvent.who);

      fcTerminate :
         toldToDie := true;

      otherwise

   end
end;
